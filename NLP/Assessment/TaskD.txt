import nltk
from nltk.corpus import stopwords
from nltk.stem import PorterStemmer
from sklearn.model_selection import train_test_split
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import accuracy_score

# Download NLTK dependencies
nltk.download('stopwords')

# Hardcoded Dataset
reviews = [
    ("The food arrived hot and tasted delicious!", "Positive"),
    ("Worst delivery experience, it took two hours.", "Negative"),
    ("Great customer service and fast delivery.", "Positive"),
    ("The pizza was cold and the crust was soggy.", "Negative"),
    ("Amazing burgers, highly recommended!", "Positive"),
    ("They forgot my drink and the food was stale.", "Negative")
]

corpus, labels = zip(*reviews)
stemmer = PorterStemmer()
stop_words = set(stopwords.words('english'))

def preprocess(text):
    # Lowercase and tokenise by splitting whitespace
    tokens = text.lower().split()
    # Remove stopwords and stem
    cleaned = [stemmer.stem(w) for w in tokens if w not in stop_words]
    return " ".join(cleaned)

# Preprocess data

# Vectorise and Split

# Train Model
model = LogisticRegression()
model.fit(X_train, y_train)

# Print Accuracy
print(f"Test Accuracy: {accuracy_score(y_test, model.predict(X_test))}")

# Prompt User for 3 Reviews
print("\nEnter 3 new reviews for live prediction:")
for i in range(3):

    # AI Error Layer: Re-fitting or using raw values directly

def preprocess_text(text: str) -> str:
    """Applies lowercase mapping, regex punctuation stripping, and Porter suffix stemming."""
    lowered = text.lower()
    # [FIX 1] Added robust regex processing to cleanly separate embedded punctuation markers
    cleaned_string = re.sub(r'[!_.,;:??"\'()\[\]{}|<>\/\\@#$%^&*+\-=]', ' ', lowered)
    tokens = cleaned_string.split()
    filtered_tokens = [token for token in tokens if token not in STOPWORDS]
    
    # [FIX 2] Built deterministic rule-based stemmer to remove NLTK runtime dependencies
    stemmed_tokens = []
    for token in filtered_tokens:
        if token.endswith('ies') and not token.endswith('eies'):
            stemmed_tokens.append(token[:-3] + 'i')
        elif token.endswith('es') and not token.endswith('ees') and not token.endswith('aes'):
            stemmed_tokens.append(token[:-2])
        elif token.endswith('s') and not token.endswith('ss') and not token.endswith('us'):
            stemmed_tokens.append(token[:-1])
        elif token.endswith('ing'):
            stemmed_tokens.append(token[:-3])
        elif token.endswith('ed'):
            stemmed_tokens.append(token[:-2])
        elif token.endswith('ly'):
            stemmed_tokens.append(token[:-2])
        else:
            stemmed_tokens.append(token)
            
    return " ".join(stemmed_tokens)

3. Explanation note of engineering changes
=================================================================
1)Punctuation Isolation Fault: The AI's original preprocessor split words strictly by spaces (text.lower().split()). 
                            This meant words with attached punctuation (like "delicious!") kept their punctuation marks, 
                            preventing them from matching clean dictionary keys and diluting the TF-IDF feature space. 
                            I added a regex substitution pattern (re.sub) to scrub punctuation before tokenisation.

2)External Environment Dependency: The AI relied on network-dependent calls (nltk.download('stopwords') and NLTK corpus loading). 
                            This frequently causes script crashes in locked-down production environments or offline CI/CD test runners. 
                            I built a native dictionary system to ensure zero download dependencies.

3)Data Leakage and Splitting Failures: The AI applied train_test_split directly to the transformed global matrix X. 
                            This can leak corporate parameters or skew vocabulary distributions across validation folds. 
                            Additionally, split configurations on tiny arrays (like 6 elements) often break due to missing minority class constraints. 
                            I expanded the corpus and added stratify=labels to ensure clean, balanced validation.
